> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Tech Stack:** Vanilla CSS, PHP-templated HTML, vanilla JS, SQLite (existing `sessions` table), Playwright Python harness.

```python
async def test_v390_shortlink_ipv4(page: Page) -> None:
    """v3.9.0 #449 — Shorten button on IPv4 mints a session and rewrites share URL to ?s=<16hex>."""
    section("v3.9.0 #449 — IPv4 short-link parity")
    await navigate(page, APP_URL)

    await submit_form(page, "#panel-ipv4 form")
    # Share bar visible after a successful calc.

    long_url = await page.locator("#panel-ipv4 .share-url").text_content()
    assert_true(f"long URL contains ip=/mask= (got {long_url})",
                "ip=" in (long_url or "") and "mask=" in (long_url or ""))
    # Click Shorten — JS POSTs to /api/v1/sessions and replaces the URL.
    async with page.expect_response(lambda r: "/api/v1/sessions" in r.url and r.request.method == "POST") as resp_info:
```

```javascript
// v3.9.0 (#449) — Shorten button mints a session-ID and replaces the share URL.
document.querySelectorAll('.share-shorten').forEach(function (btn) {
    btn.addEventListener('click', async function () {
        const tab  = btn.dataset.tab  || 'ipv4';
        const ip   = btn.dataset.ip   || '';
        const mask = btn.dataset.mask || '';
        if (!ip) return;
        btn.disabled = true;
        try {
            const resp = await fetch('api/v1/sessions', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ type: 'calc', tab: tab, ip: ip, mask: mask }),
            });
            if (!resp.ok) { showToast('Shorten failed'); return; }
            const env = await resp.json();
            const sid = (env.data && env.data.id) || env.id;
            if (!sid) { showToast('Shorten failed'); return; }
            const shortPath = window.location.pathname + '?s=' + encodeURIComponent(sid);
            const shortAbs  = window.location.origin + shortPath;
            const bar = btn.closest('.share-bar');
            if (bar) {
                const urlEl  = bar.querySelector('.share-url');
                const copyEl = bar.querySelector('.share-copy');
                if (urlEl)  urlEl.textContent = shortAbs;
                if (copyEl) copyEl.setAttribute('data-copy', shortAbs);
            }
            btn.style.display = 'none';
        } catch (e) {
            showToast('Shorten failed');
        } finally {
            btn.disabled = false;
        }
    });
});
```
