**Goal:** Ship v3.4.0 — close v3.3.0's six carry-forward debt items, harden the IPv6 surface (a11y, bulk endpoint, type-detection breakout), introduce per-tool URL routes (`/ipv6/derive`-style) without breaking existing query-param URLs, and ship two parity tools (rdns6, mapped6/NAT64).

**Architecture:** Sequential PR fan-out off `release/v3.4.0`, modelled on v3.3.0's PR1–PR7. Each item is its own PR; all merge into `release/v3.4.0`; final release-cut PR opens `release/v3.4.0 → main`. Refactor PRs land before new tools so new tools ship with the post-refactor patterns from day one.

**Tech Stack:** PHP 8.2+ (strict types, GMP), SQLite3, plain JS, Apache `.htaccess` rewrites, mkdocs + spectral, PHPUnit 11, Playwright 1.4x via docker, PHPStan L9, PHPCS PSR-12, Semgrep.

```php
/**
 * Render a single copy-to-clipboard button with safe escaping.
 *
 * Mirrors the existing inline pattern (`<button class="copy-btn"
 * data-clipboard-text="...">`) used throughout templates/layout.php.
 * The JS handler in app.js binds via the `.copy-btn` selector and
 * reads `data-clipboard-text`, so existing JS continues to work
 * unchanged.
 *
 * @param string $value The literal text that will be written to the
 *                      clipboard when the button is clicked.
 * @param string $label Accessible label (also used as button text or
 *                      tooltip per existing convention).
 */
function copy_button(string $value, string $label): string
{
    $escapedValue = htmlspecialchars($value, ENT_QUOTES | ENT_SUBSTITUTE, 'UTF-8');
    $escapedLabel = htmlspecialchars($label, ENT_QUOTES | ENT_SUBSTITUTE, 'UTF-8');
    return sprintf(
        '<button type="button" class="copy-btn" data-clipboard-text="%s" aria-label="%s">%s</button>',
        $escapedValue,
        $escapedLabel,
        $escapedLabel
    );
}
```

(Adjust the inner HTML structure if the audit in Step 2 reveals an icon `<svg>` or different inner content — the helper must reproduce that pattern exactly.)

```php
<?php
declare(strict_types=1);

/**
 * IPv6 reverse-DNS helpers (RFC 3596).
 *
 * Generates ip6.arpa zone names from an IPv6 address and optional
 * prefix length. Prefix length must be a multiple of 4 — the RFC's
 * nibble-boundary requirement for delegation.
 */

/**
 * @throws InvalidArgumentException if $address is not a valid IPv6
 *         literal, or $prefix is not in [0, 128] or not nibble-aligned.
 */
function ipv6_to_arpa(string $address, ?int $prefix = null): string
{
    $bin = @inet_pton($address);
    if ($bin === false || strlen($bin) !== 16) {
        throw new InvalidArgumentException('Invalid IPv6 address');
    }

    $prefix = $prefix ?? 128;
    if ($prefix < 0 || $prefix > 128) {
        throw new InvalidArgumentException('Prefix length must be 0..128');
    }
    if ($prefix % 4 !== 0) {
        throw new InvalidArgumentException(
            'Prefix length must be a multiple of 4 for ip6.arpa zone delegation'
        );
    }

    $hex = bin2hex($bin); // 32 nibbles
    $nibbles = $prefix / 4;
    $significant = substr($hex, 0, $nibbles);
    $reversed = strrev($significant);
    $dotted = implode('.', str_split($reversed));
    return $dotted === '' ? 'ip6.arpa' : $dotted . '.ip6.arpa';
}
```

- [ ] **Step 3: Implement `functions-mapped6.php`**

```php
<?php
declare(strict_types=1);

const MAPPED6_PREFIX_BIN = "\x00\x00\x00\x00\x00\x00\x00\x00\x00\x00\xff\xff";
const NAT64_DEFAULT_PREFIX = '64:ff9b::/96';

function is_ipv4_mapped(string $address): bool
{
    $bin = @inet_pton($address);
    return $bin !== false
        && strlen($bin) === 16
        && substr($bin, 0, 12) === MAPPED6_PREFIX_BIN;
}

function ipv4_to_mapped(string $v4): string
{
    $bin = @inet_pton($v4);
    if ($bin === false || strlen($bin) !== 4) {
        throw new InvalidArgumentException('Invalid IPv4 address');
    }
    $v6bin = MAPPED6_PREFIX_BIN . $bin;
    // inet_ntop renders ::ffff:a.b.c.d for v4-mapped addresses
    return inet_ntop($v6bin);
}

function mapped_to_ipv4(string $v6): string
{
    if (!is_ipv4_mapped($v6)) {
        throw new InvalidArgumentException('Not an IPv4-mapped IPv6 address');
    }
    $bin = inet_pton($v6);
    return inet_ntop(substr($bin, 12, 4));
}

/**
 * @throws InvalidArgumentException if $prefix is not a /96 NAT64 prefix.
 */
function _nat64_prefix_bin(string $prefix): string
{
    if (!preg_match('#^(.+)/96$#', $prefix, $m)) {
        throw new InvalidArgumentException('NAT64 prefix must be /96');
    }
    $bin = @inet_pton($m[1]);
    if ($bin === false || strlen($bin) !== 16) {
        throw new InvalidArgumentException('Invalid NAT64 prefix address');
    }
    return substr($bin, 0, 12);
}

function is_nat64(string $address, string $prefix = NAT64_DEFAULT_PREFIX): bool
{
    $bin = @inet_pton($address);
    if ($bin === false || strlen($bin) !== 16) {
        return false;
    }
    return substr($bin, 0, 12) === _nat64_prefix_bin($prefix);
}

function ipv4_to_nat64(string $v4, string $prefix = NAT64_DEFAULT_PREFIX): string
{
    $v4bin = @inet_pton($v4);
    if ($v4bin === false || strlen($v4bin) !== 4) {
        throw new InvalidArgumentException('Invalid IPv4 address');
    }
    $prefBin = _nat64_prefix_bin($prefix);
    return inet_ntop($prefBin . $v4bin);
}

function nat64_to_ipv4(string $v6, string $prefix = NAT64_DEFAULT_PREFIX): string
{
    if (!is_nat64($v6, $prefix)) {
        throw new InvalidArgumentException('Address is not within the supplied NAT64 prefix');
    }
    $bin = inet_pton($v6);
    return inet_ntop(substr($bin, 12, 4));
}
```

```php
<?php
declare(strict_types=1);

require_once __DIR__ . '/../helpers.php';
require_once __DIR__ . '/../../../includes/functions-mapped6.php';

function handle_mapped6(array $body): void
{
    $input = $body['input'] ?? null;
    $prefix = $body['nat64_prefix'] ?? NAT64_DEFAULT_PREFIX;

    if (!is_string($input) || $input === '') {
        json_err('input (string) required — IPv4 or v4-mapped/NAT64 IPv6', 400);
    }

    try {
        // Detect input form, return all four representations
        if (filter_var($input, FILTER_VALIDATE_IP, FILTER_FLAG_IPV4)) {
            $v4     = $input;
            $mapped = ipv4_to_mapped($v4);
            $nat64  = ipv4_to_nat64($v4, $prefix);
        } elseif (is_ipv4_mapped($input)) {
            $v4     = mapped_to_ipv4($input);
            $mapped = $input;
            $nat64  = ipv4_to_nat64($v4, $prefix);
        } elseif (is_nat64($input, $prefix)) {
            $v4     = nat64_to_ipv4($input, $prefix);
            $mapped = ipv4_to_mapped($v4);
            $nat64  = $input;
        } else {
            json_err('Input is not IPv4, IPv4-mapped IPv6, or NAT64 IPv6 within the supplied prefix', 400);
        }

        json_ok([
            'input'         => $input,
            'ipv4'          => $v4,
            'ipv4_mapped'   => $mapped,
            'nat64'         => $nat64,
            'nat64_prefix'  => $prefix,
        ]);
    } catch (InvalidArgumentException $e) {
        json_err($e->getMessage(), 400);
    }
}
```
